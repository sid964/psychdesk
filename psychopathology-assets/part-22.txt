function searchItems(term){let q=normSearch(term);if(!q)return[];const items=[];
 for(const c of CHAPTERS){
  if(normSearch(c.title).includes(q))items.push({kind:'Chapter',title:c.title,page:c.printedStart,partId:c.modules[0].readingPartId});
  for(const m of c.modules){
   const p=PARTS.find(p=>p.id===m.readingPartId);
   if(normSearch(m.title+' '+m.number).includes(q))items.push({kind:'Notes '+m.number,title:m.title,page:m.printedStart,partId:m.readingPartId});
   for(const s of (p?.studySections||[])) if(normSearch(s.title+' '+s.text).includes(q))items.push({kind:'Study explanation · '+m.number,title:s.title,page:s.sourcePrintedStart,partId:m.readingPartId,sectionId:s.id});
   for(const h of m.headings){const deep=(p?.deepTopics||[]).find(t=>t.id===h.id);if(normSearch(h.title+' '+(p?.classroomNotes||[]).filter(x=>x.headingId===h.id).map(x=>x.explanation+' '+(x.followUpExplanation||'')+' '+x.bullets.map(b=>b.label+' '+b.explanation).join(' ')).join(' ')+' '+(deep?.facts||[]).map(f=>f.text).join(' ')).includes(q))items.push({kind:(deep?.facts.length?'Detailed topic notes · ':'Textbook topic · ')+m.number,title:h.title,page:h.printedPage,partId:m.readingPartId,topicId:h.id});}
  }
 }
 for(const a of BOOK.appendices){if(normSearch(a.title).includes(q))items.push({kind:a.label,title:a.title,page:a.printedStart,pdfPage:a.pdfStart});for(const h of a.headings)if(normSearch(h.title).includes(q))items.push({kind:a.label,title:h.title,page:h.printedPage,pdfPage:h.pdfPage})}
 for(const a of BOOK.backMatter)if(normSearch(a.title).includes(q))items.push({kind:'Back matter',title:a.title,page:a.printedStart,pdfPage:a.pdfStart});
 return items;
}
